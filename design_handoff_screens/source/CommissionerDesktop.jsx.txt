function ResultOverrides() {

    <div style={{ fontSize: 14, color: "var(--muted-foreground)", maxWidth: 720 }}>Scores come from CollegeFootballData every few minutes. Override only when the feed is wrong or a game will not be played. Voiding a game scores zero for everyone and drops any Lock on it.</div>
    <Card style={{ padding: 0, gap: 0, overflow: "hidden" }}>
      <Table>
        <TableHeader><TableRow><TableHead style={{ paddingLeft: 12 }}>Game</TableHead><TableHead>Kickoff</TableHead><TableHead style={NUM}>Away</TableHead><TableHead style={NUM}>Home</TableHead><TableHead>Status</TableHead><TableHead style={{ textAlign: "right", paddingRight: 12 }}>Override</TableHead></TableRow></TableHeader>
        <TableBody>{games.map(g => { const ed = editing === g.id; const voided = g.status === "void"; return <TableRow key={g.id} style={{ height: 56, opacity: voided ? .7 : 1, background: ed ? "var(--accent)" : "transparent" }}>
          <TableCell style={{ paddingLeft: 12, ...DISPLAY, fontSize: 16 }}><span style={{ display: "inline-flex", alignItems: "center", gap: 8 }}><TeamLogo team={g.away} size={22} />{g.away} <span style={{ fontFamily: "var(--font-sans)", fontWeight: 500, color: "var(--muted-foreground)" }}>at</span> <TeamLogo team={g.home} size={22} />{g.home}{g.tiebreaker && <Badge variant="secondary" style={{ marginLeft: 8 }}>Tiebreaker</Badge>}</span></TableCell>
          <TableCell style={{ color: "var(--muted-foreground)" }}>{g.time}</TableCell>
          <TableCell style={NUM}>{ed ? <Input defaultValue={g.awayScore ?? ""} inputMode="numeric" style={{ width: 72, textAlign: "right" }} onChange={e => set(g.id, { awayScore: +e.target.value })} /> : <span style={{ ...DISPLAY, fontSize: 18 }}>{g.awayScore ?? "–"}</span>}</TableCell>
          <TableCell style={NUM}>{ed ? <Input defaultValue={g.homeScore ?? ""} inputMode="numeric" style={{ width: 72, textAlign: "right" }} onChange={e => set(g.id, { homeScore: +e.target.value })} /> : <span style={{ ...DISPLAY, fontSize: 18 }}>{g.homeScore ?? "–"}</span>}</TableCell>
          <TableCell>{voided ? <Badge variant="outline">Void</Badge> : g.status === "live" ? <Badge variant="live"><Icon name="radio" size={12} />{g.clock}</Badge> : g.status === "final" ? <Badge variant="default">Final{g.overridden ?           <TableCell style={{ textAlign: "right", paddingRight: 12 }}><span style={{ display: "inline-flex", gap: 4 }}>
            {voided ? <Button variant="outline" size="sm" onClick={() => set(g.id, { status: "scheduled", time: "Sat 7:30 PM" })}><Icon name="undo-2" />Restore game</Button>
            : ed ? <><Button size="sm" onClick={() => { set(g.id, { status: "final", overridden: true, clock: undefined }); setEditing(null); }}><Icon name="check" />Save as Final</Button><Button variant="ghost" size="sm" onClick={() => setEditing(null)}>Cancel</Button></>

          </span></TableCell>

      </Table>
    </Card>
  </div>;
}
