// Preview-only theme switch, mounted outside the phone frame so it never covers app content
function ThemeSwitch(){
  const [dark,setDark]=React.useState(new URLSearchParams(location.search).get("theme")==="dark");
  React.useEffect(()=>{document.documentElement.classList.toggle("dark",dark)},[dark]);
  // Only render when the backdrop beside the frame is wide enough to hold it — at the card's own
  // 390px width the frame fills the window, and any window-anchored control would sit on the app.
  const [gutter,setGutter]=React.useState(0);
  React.useEffect(()=>{
    const host=document.getElementById("preview-controls");
    const measure=()=>{const r=document.getElementById("root").getBoundingClientRect();const g=Math.min(r.left,window.innerWidth-r.right);setGutter(g);host.style.left=Math.max(10,r.left-42)+"px"};
    measure(); window.addEventListener("resize",measure); return ()=>window.removeEventListener("resize",measure);
  },[]);
  if(gutter<44) return null;
  return <button onClick={()=>setDark(d=>!d)} aria-label="Toggle dark theme" title="Preview control, not part of the app" style={{width:32,height:32,minHeight:32,borderRadius:9999,border:"1px solid var(--border)",background:"var(--card)",color:"var(--muted-foreground)",display:"grid",placeItems:"center",cursor:"pointer",boxShadow:"var(--shadow-sm)"}}><Icon name={dark?"sun":"moon"} size={16}/></button>;
}
ReactDOM.createRoot(document.getElementById("preview-controls")).render(<ThemeSwitch/>);
